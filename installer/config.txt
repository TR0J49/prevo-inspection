# Address of the audit server. Must be the NETWORK ip, not localhost,
# because the workstation has to reach it across the LAN.
#
# IMPORTANT: give the server a fixed address first (DHCP reservation on
# the router, and a wired connection). If the server's ip changes, every
# installed PC stops reporting and has to be visited again.
SERVER_URL=http://192.168.1.26:8000

# How often the audit runs automatically, in hours.
# The audit also runs once at every startup, regardless of this value.
INTERVAL_HOURS=3

# Maximum random delay in seconds before each scheduled run.
# Stops every PC in a branch hitting the server at the same moment.
# 300 = spread the load over 5 minutes. Set 0 to disable.
JITTER_SECONDS=300
